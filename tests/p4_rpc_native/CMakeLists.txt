cmake_minimum_required(VERSION 3.28)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(p4_rpc_native C)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(src "${repo}/boards/waveshare_esp32p4_wifi6/zephyr/src")
if(NOT P4_RPC_GENERATED OR NOT TARGET modules__nanopb)
  message(FATAL_ERROR "Pass the existing pinned station RPC generated directory")
endif()
target_compile_options(modules__nanopb PRIVATE "SHELL:-include ${src}/rpc_allocator.h")
target_include_directories(app PRIVATE "${src}" "${P4_RPC_GENERATED}" "${ZEPHYR_BASE}/drivers/misc/esp_hosted_mcu")
target_include_directories(app PRIVATE "${repo}/common/storage")
target_compile_definitions(app PRIVATE P4_RPC_ALLOC_TEST=1)
target_sources(app PRIVATE main.c driver.c "${src}/rpc_allocator.c" "${src}/rpc_decode.c" "${P4_RPC_GENERATED}/esp_hosted_mcu_rpc.pb.c")
target_sources(app PRIVATE "${repo}/common/storage/storage.c")
